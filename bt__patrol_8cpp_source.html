44static geometry_msgs::msg::PoseStamped makePose(
45 const rclcpp::Node::SharedPtr & node, const Waypoint & wp)
46{
47 geometry_msgs::msg::PoseStamped pose;
48 pose.header.frame_id = "map";
49 pose.header.stamp = node->now();
50 pose.pose.position.x = wp.x;
51 pose.pose.position.y = wp.y;
52 pose.pose.orientation.z = std::sin(wp.theta / 2.0);
53 pose.pose.orientation.w = std::cos(wp.theta / 2.0);
54 return pose;
55}

108class NavigateToWaypoint : public BT::StatefulActionNode
109{
110public:
111 using NavigateToPose = nav2_msgs::action::NavigateToPose;
112 using GoalHandle = rclcpp_action::ClientGoalHandle<NavigateToPose>;
113
114 NavigateToWaypoint(const std::string & name, const BT::NodeConfig & config,
115 std::shared_ptr<PatrolContext> context)
116 : BT::StatefulActionNode(name, config), context_(std::move(context)) {}
117
118 BT::NodeStatus onStart() override
119 {
120 if (!context_->current) {
121 return BT::NodeStatus::FAILURE;
122 }
123 NavigateToPose::Goal goal;
124 goal.pose = makePose(context_->node, *context_->current);
125 RCLCPP_INFO(
126 context_->node->get_logger(), "%s: sending goal to %s", name().c_str(),
127 context_->current->name.c_str());
128 goal_future_ = context_->nav_client->async_send_goal(goal);
129 result_requested_ = false;
130 return BT::NodeStatus::RUNNING;
131 }
132
133 BT::NodeStatus onRunning() override
134 {
135 if (!goal_handle_) {
136 if (goal_future_.valid() && goal_future_.wait_for(0ms) == std::future_status::ready) {
137 goal_handle_ = goal_future_.get();
138 if (!goal_handle_) {
139 RCLCPP_WARN(context_->node->get_logger(), "%s: goal rejected", name().c_str());
140 return BT::NodeStatus::FAILURE;
141 }
142 } else {
143 return BT::NodeStatus::RUNNING;
144 }
145 }
146
147 if (!result_requested_) {
148 result_future_ = context_->nav_client->async_get_result(goal_handle_);
149 result_requested_ = true;
150 return BT::NodeStatus::RUNNING;
151 }
152
153 if (result_future_.valid() && result_future_.wait_for(0ms) == std::future_status::ready) {
154 const auto result = result_future_.get();
155 goal_handle_.reset();
156 result_requested_ = false;
157 if (result.code == rclcpp_action::ResultCode::SUCCEEDED) {
158 RCLCPP_INFO(context_->node->get_logger(), "%s: navigation succeeded", name().c_str());
159 return BT::NodeStatus::SUCCESS;
160 }
161 RCLCPP_WARN(context_->node->get_logger(), "%s: navigation failed", name().c_str());
162 return BT::NodeStatus::FAILURE;
163 }
164
165 return BT::NodeStatus::RUNNING;
166 }
167
168 void onHalted() override
169 {
170 if (goal_handle_) {
171 context_->nav_client->async_cancel_goal(goal_handle_);
172 }
173 goal_handle_.reset();
174 result_requested_ = false;
175 }
176
177private:
178 std::shared_ptr<PatrolContext> context_;
179 std::shared_future<GoalHandle::SharedPtr> goal_future_;
180 GoalHandle::SharedPtr goal_handle_;
181 std::shared_future<GoalHandle::WrappedResult> result_future_;
182 bool result_requested_{false};
183};

321 auto bt_xml = valueArg(argc, argv, "--bt-xml=", default_xml);
322 if (bt_xml.empty()) {
323 bt_xml = default_xml;
324 }
325 const int groot_port = std::stoi(valueArg(argc, argv, "--groot-port=", "1667"));
326
327 auto context = std::make_shared<PatrolContext>();
328 context->node = rclcpp::Node::make_shared("bt_patrol");
329 context->nav_client = rclcpp_action::create_client<nav2_msgs::action::NavigateToPose>(
330 context->node, "navigate_to_pose");
331 context->event_pub = context->node->create_publisher<std_msgs::msg::String>("patrol_events", 10);
332 context->cmd_vel_pub = context->node->create_publisher<geometry_msgs::msg::Twist>("cmd_vel", 10);
333 context->skip_errors = skip_errors;
334 context->queue = loadWaypoints();
335
336 if (random_order) {
337 std::random_device rd;
338 std::mt19937 gen(rd());
339 std::shuffle(context->queue.begin(), context->queue.end(), gen);
340 }
341
342 RCLCPP_INFO(context->node->get_logger(), "Custom BehaviorTree.CPP patrol starting");
343 RCLCPP_INFO(context->node->get_logger(), "BT XML: %s", bt_xml.c_str());
344 RCLCPP_INFO(context->node->get_logger(), "Groot2 live publisher port: %d", groot_port);
345
346 BT::BehaviorTreeFactory factory;
347 factory.registerBuilder<SelectWaypoint>(
348 "SelectWaypoint", [context](const std::string & name, const BT::NodeConfig & config) {
349 return std::make_unique<SelectWaypoint>(name, config, context);
350 });
351 factory.registerBuilder<NavigateToWaypoint>(
352 "NavigateToWaypoint", [context](const std::string & name, const BT::NodeConfig & config) {
353 return std::make_unique<NavigateToWaypoint>(name, config, context);
354 });
355 factory.registerBuilder<BackUpRecovery>(
356 "BackUpRecovery", [context](const std::string & name, const BT::NodeConfig & config) {
357 return std::make_unique<BackUpRecovery>(name, config, context);
358 });
359 factory.registerBuilder<PublishArrival>(
360 "PublishArrival", [context](const std::string & name, const BT::NodeConfig & config) {
361 return std::make_unique<PublishArrival>(name, config, context);
362 });
363 factory.registerBuilder<HandleFailure>(
364 "HandleFailure", [context](const std::string & name, const BT::NodeConfig & config) {
365 return std::make_unique<HandleFailure>(name, config, context);
366 });
367 factory.registerBuilder<PatrolLoop>(
368 "PatrolLoop", [context](const std::string & name, const BT::NodeConfig & config) {
369 return std::make_unique<PatrolLoop>(name, config, context);
370 });
371
372 auto tree = factory.createTreeFromFile(bt_xml);
373 std::unique_ptr<BT::Groot2Publisher> groot_publisher;
374 try {
375 groot_publisher = std::make_unique<BT::Groot2Publisher>(
376 tree, static_cast<unsigned>(groot_port));
377 } catch (const std::exception & ex) {
378 RCLCPP_WARN(
379 context->node->get_logger(), "Could not start Groot2 publisher on port %d: %s",
380 groot_port, ex.what());
381 }
382
383 if (!context->nav_client->wait_for_action_server(60s)) {
384 RCLCPP_ERROR(context->node->get_logger(), "Timeout waiting for navigate_to_pose action server");
385 rclcpp::shutdown();
386 return 1;
387 }
388
389 rclcpp::Rate rate(5.0);
390 BT::NodeStatus status = BT::NodeStatus::IDLE;
391 while (rclcpp::ok()) {
392 rclcpp::spin_some(context->node);
393 status = tree.tickOnce();
394 if (status == BT::NodeStatus::SUCCESS || status == BT::NodeStatus::FAILURE) {
395 break;
396 }
397 rate.sleep();
398 }
399
400 tree.haltTree();
401 context->cmd_vel_pub->publish(geometry_msgs::msg::Twist());
402 RCLCPP_INFO(context->node->get_logger(), "BT finished with status: %s", BT::toStr(status).c_str());
403 rclcpp::shutdown();
404 return status == BT::NodeStatus::FAILURE ? 1 : 0;
