43def load_robot_speed() -> float:
44 """Read max linear velocity from Nav2 params (m/s)."""
45 candidates = [
46 os.path.join(_project_root(), 'ros2_ws/src/home_robot/config/nav2_params_slam.yaml'),
47 os.path.join(_project_root(), 'ros2_ws/src/home_robot/config/nav2_params.yaml'),
48 ]
49 for path in candidates:
50 if not os.path.exists(path):
51 continue
52 try:
53 with open(path, 'r') as stream:
54 cfg = yaml.safe_load(stream)
55 max_vel = cfg.get('velocity_smoother', {}).get('ros__parameters', {}).get('max_velocity')
56 if isinstance(max_vel, list) and len(max_vel) >= 1:
57 speed = float(max_vel[0])

69def sequential_links(names: List[str]) -> List[Tuple[str, str]]:
70 links = []
71 for idx in range(len(names) - 1):
72 links.append((names[idx], names[idx + 1]))

81def nearest_waypoint(waypoints: Dict[str, dict], x: float, y: float) -> str:
82 best_name = None
83 best_dist = None
84 for name, wp in waypoints.items():
85 dist = math.hypot(float(wp['x']) - x, float(wp['y']) - y)
86 if best_dist is None or dist < best_dist:
87 best_name = sanitize(name)
88 best_dist = dist
89 return best_name

92def resolve_start(waypoints: Dict[str, dict], start: str = 'auto', tf_timeout: float = 5.0) -> str:
93 names = [sanitize(name) for name in waypoints.keys()]
94 if start != 'auto':
95 clean = sanitize(start)
96 if clean not in names:
97 raise ValueError(f'Unknown start waypoint: {start}')
98 return clean
99
100 if rclpy is None:
101 return 'inicio' if 'inicio' in names else names[0]
102
103 did_init = False
104 if not rclpy.ok():
105 rclpy.init(args=None)
106 did_init = True
107 node = rclpy.create_node('patrol_problem_start_resolver')
108 tf_buffer = Buffer()
109 TransformListener(tf_buffer, node)
110 deadline = node.get_clock().now() + Duration(seconds=tf_timeout)
111
112 try:
113 while rclpy.ok() and node.get_clock().now() < deadline:
114 rclpy.spin_once(node, timeout_sec=0.1)
115 try:
116 transform = tf_buffer.lookup_transform('map', 'base_link', Time())
117 x = transform.transform.translation.x
118 y = transform.transform.translation.y
119 start_wp = nearest_waypoint(waypoints, x, y)
120 node.get_logger().info(
121 f'Auto PDDL start: robot at ({x:.2f}, {y:.2f}) -> {start_wp}')
122 return start_wp
123 except Exception:
124 pass
125 node.get_logger().warn('Could not resolve map->base_link; falling back to inicio')
126 finally:
127 node.destroy_node()
128 if did_init:
129 rclpy.shutdown()
130
131 return 'inicio' if 'inicio' in names else names[0]

134def _travel_time(waypoints: Dict[str, dict], src: str, dst: str, speed: float) -> float:
135 s = waypoints.get(src, {})
136 d = waypoints.get(dst, {})
137 dist = math.hypot(float(s.get('x', 0)) - float(d.get('x', 0)),
138 float(s.get('y', 0)) - float(d.get('y', 0)))
139 return dist / speed if speed > 0 else 5.0

142def generate_problem(
143 links_mode: str = 'all_to_all',
144 start: str = 'auto',
145 tf_timeout: float = 5.0,
146 robot_speed: float = None,
147 visited_names=None,
148 blocked: bool = False,
149) -> str:
150 if robot_speed is None or robot_speed <= 0:
151 robot_speed = load_robot_speed()
152 waypoints = load_waypoints()
153 names = [sanitize(name) for name in waypoints.keys()]
154 start = resolve_start(waypoints, start, tf_timeout)
155 links = all_to_all_links(names) if links_mode == 'all_to_all' else sequential_links(names)
156
157 objects = ' '.join(names)
158 init_lines = [f' (robot_at robot1 {start})']
159 already_visited = set(visited_names) if visited_names else set()
160 already_visited.add(start)
161 for v in sorted(already_visited):
162 if v in names:
163 init_lines.append(f' (visited {v})')
164 init_lines.extend(f' (connected {src} {dst})' for src, dst in links)
165 for src, dst in links:
166 t = _travel_time(waypoints, src, dst, robot_speed)
167 init_lines.append(f' (= (travel_time {src} {dst}) {t:.2f})')
168 if blocked:
169 init_lines.append(' (blocked)')
170 goal_lines = [f' (visited {name})' for name in names]
171 goal_lines.append(f' (robot_at robot1 inicio)')
172
173 return f"""(define (problem home_robot_patrol_problem)
174 (:domain home_robot_patrol)
175
176 (:objects
177 robot1 - robot
178 {objects} - location
179 )
180
181 (:init
182{os.linesep.join(init_lines)}
183 )
184
185 (:goal
186 (and
187{os.linesep.join(goal_lines)}
188 )
189 )
190)
191"""

194def generate_terminal_commands(
195 links_mode: str = 'all_to_all', start: str = 'auto', tf_timeout: float = 5.0
196) -> str:
197 waypoints = load_waypoints()
198 names = [sanitize(name) for name in waypoints.keys()]
199 start = resolve_start(waypoints, start, tf_timeout)
200 links = all_to_all_links(names) if links_mode == 'all_to_all' else sequential_links(names)
201
202 lines = ['set instance robot1 robot']
203 lines.extend(f'set instance {name} location' for name in names)
204 lines.append(f'set predicate (robot_at robot1 {start})')
205 lines.append(f'set predicate (visited {start})')
206 lines.extend(f'set predicate (connected {src} {dst})' for src, dst in links)
207 # plansys2_terminal is picky with compound expression spacing. The compact
208 # form matches the official examples: set goal (and(predicate)(predicate))
209 goal = '(and' + ''.join(f'(visited {name})' for name in names) + ')'
210 lines.append(f'set goal {goal}')
211 lines.append('run')
212 return os.linesep.join(lines) + os.linesep
