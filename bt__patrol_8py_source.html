48class Sequence(Behavior):
49 def __init__(self, name: str, children: list[Behavior]):
50 super().__init__(name)
51 self.children = children
52 self.index = 0
53
54 def tick(self, blackboard: dict) -> Status:
55 while self.index < len(self.children):
56 status = self.children[self.index].tick(blackboard)
57 if status == Status.RUNNING:
58 return Status.RUNNING
59 if status == Status.FAILURE:
60 self.reset()
61 return Status.FAILURE
62 self.children[self.index].reset()
63 self.index += 1
64 self.reset()
65 return Status.SUCCESS
66
67 def reset(self) -> None:
68 self.index = 0
69 for child in self.children:
70 child.reset()

98class PatrolLoop(Behavior):
99 def __init__(self, child: Behavior):
100 super().__init__('PatrolLoop')
101 self.child = child
102
103 def tick(self, blackboard: dict) -> Status:
104 if blackboard.get('patrol_done', False):
105 return Status.SUCCESS
106 status = self.child.tick(blackboard)
107 if status == Status.SUCCESS:
108 self.child.reset()
109 return Status.RUNNING
110 if status == Status.FAILURE:
111 if blackboard.get('patrol_done', False):
112 return Status.SUCCESS
113 return Status.FAILURE
114 return Status.RUNNING
115
116 def reset(self) -> None:
117 self.child.reset()

120class SelectWaypoint(Behavior):
121 def __init__(self, navigator: BasicNavigator):
122 super().__init__('SelectWaypoint')
123 self.navigator = navigator
124
125 def tick(self, blackboard: dict) -> Status:
126 queue = blackboard['wp_queue']
127 if not queue:
128 blackboard['patrol_done'] = True
129 self.navigator.get_logger().info('BT patrol complete: no waypoints left')
130 return Status.FAILURE
131
132 name, wp = queue.pop(0)
133 blackboard['current_wp_name'] = name
134 blackboard['current_wp'] = wp
135 blackboard['navigation_failed'] = False
136 total = blackboard['wp_total']
137 current = total - len(queue)
138 self.navigator.get_logger().info(f'[{current}/{total}] BT selected waypoint: {name}')
139 return Status.SUCCESS

142class NavigateToWaypoint(Behavior):
143 def __init__(self, navigator: BasicNavigator, label: str,
144 stuck_timeout: float = 15.0, stuck_dist_threshold: float = 0.1):
145 super().__init__(label)
146 self.navigator = navigator
147 self.started = False
148 self.stuck_timeout = stuck_timeout
149 self.stuck_dist_threshold = stuck_dist_threshold
150 self.last_dist = None
151 self.last_dist_time = None
152
153 def tick(self, blackboard: dict) -> Status:
154 name = blackboard['current_wp_name']
155 if not self.started:
156 pose = make_pose(self.navigator, blackboard['current_wp'])
157 self.navigator.get_logger().info(f'{self.name}: sending Nav2 goal for {name}')
158 self.navigator.goToPose(pose)
159 self.started = True
160 self.last_dist = None
161 self.last_dist_time = None
162 return Status.RUNNING
163
164 feedback = self.navigator.getFeedback()
165 now = time.time()
166 if feedback and feedback.distance_remaining is not None:
167 dist = feedback.distance_remaining
168 if self.last_dist is not None:
169 if abs(dist - self.last_dist) <= self.stuck_dist_threshold:
170 if now - self.last_dist_time >= self.stuck_timeout:
171 self.navigator.get_logger().warn(
172 f'{self.name}: STUCK detected ({self.stuck_timeout}s '
173 f'without progress). Aborting navigation.')
174 self.navigator.cancelTask()
175 blackboard['navigation_failed'] = True
176 blackboard['is_stuck'] = True
177 self.reset()
178 return Status.FAILURE
179 else:
180 self.last_dist = dist
181 self.last_dist_time = now
182 else:
183 self.last_dist = dist
184 self.last_dist_time = now
185
186 if dist > 0:
187 self.navigator.get_logger().info(f'{name}: {dist:.2f} m remaining')
188
189 if not self.navigator.isTaskComplete():
190 return Status.RUNNING
191
192 result = self.navigator.getResult()
193 if result == TaskResult.SUCCEEDED:
194 self.navigator.get_logger().info(f'{self.name}: reached {name}')
195 return Status.SUCCESS
196
197 blackboard['navigation_failed'] = True
198 self.navigator.get_logger().warn(f'{self.name}: failed to reach {name}')
199 return Status.FAILURE
200
201 def reset(self) -> None:
202 self.started = False
203 self.last_dist = None
204 self.last_dist_time = None

284def make_pose(navigator: BasicNavigator, wp: dict) -> PoseStamped:
285 pose = PoseStamped()
286 pose.header.frame_id = 'map'
287 pose.header.stamp = navigator.get_clock().now().to_msg()
288 pose.pose.position.x = float(wp['x'])
289 pose.pose.position.y = float(wp['y'])
290 theta = float(wp['theta'])
291 pose.pose.orientation.z = math.sin(theta / 2.0)
292 pose.pose.orientation.w = math.cos(theta / 2.0)
293 return pose

296def wait_for_nav2_action(navigator: BasicNavigator) -> bool:
297 action_client = ActionClient(navigator, NavigateToPose, 'navigate_to_pose')
298 start = time.time()

309def build_tree(navigator: BasicNavigator, event_pub, cmd_vel_pub) -> Behavior:
310 return PatrolLoop(
311 Sequence('PatrolOneWaypoint', [
312 SelectWaypoint(navigator),
313 Fallback('NavigateOrRecover', [
314 Sequence('PrimaryNavigation', [

329def main() -> None:
330 global stop_patrol
331
332 signal.signal(signal.SIGTERM, signal_handler)
333 signal.signal(signal.SIGINT, signal_handler)
334
335 use_random = '--random' in sys.argv
336 skip_errors = '--skip-errors' in sys.argv
337
338 rclpy.init()
339 navigator = BasicNavigator()
340 event_pub = navigator.create_publisher(String, 'patrol_events', 10)
341 cmd_vel_pub = navigator.create_publisher(Twist, '/cmd_vel', 10)
342
343 def stop_callback(msg):
344 del msg
345 global stop_patrol
346 navigator.get_logger().warn('Stop request received on /stop_patrol')
347 stop_patrol = True
348
349 navigator.create_subscription(String, '/stop_patrol', stop_callback, 10)
350
351 try:
352 waypoints = load_waypoints()
353 except FileNotFoundError as exc:
354 navigator.get_logger().error(str(exc))
355 rclpy.shutdown()
356 return
357
358 wp_items = list(waypoints.items())
359 if use_random:
360 random.shuffle(wp_items)
361
362 blackboard = {
363 'wp_queue': wp_items,
364 'wp_total': len(wp_items),
365 'skip_errors': skip_errors,
366 'patrol_done': False,
367 }
368
369 navigator.get_logger().info('Custom BT patrol starting')
370 navigator.get_logger().info(f'Mode: random={use_random}, skip_errors={skip_errors}')
371
372 if not wait_for_nav2_action(navigator):
373 rclpy.shutdown()
374 return
375
376 tree = build_tree(navigator, event_pub, cmd_vel_pub)
377
378 try:
379 while rclpy.ok() and not stop_patrol:
380 rclpy.spin_once(navigator, timeout_sec=0.1)
381 status = tree.tick(blackboard)
382 navigator.get_logger().info(f'BT root status: {status.value}')
383 if status in (Status.SUCCESS, Status.FAILURE):
384 break
385 time.sleep(0.2)
386 finally:
387 if rclpy.ok():
388 navigator.cancelTask()
389 cmd_vel_pub.publish(Twist())
390 navigator.destroy_node()
391 rclpy.shutdown()
392
393
394if __name__ == '__main__':
395 main()
