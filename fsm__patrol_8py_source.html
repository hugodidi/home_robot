1#!/usr/bin/env python3
2"""
3FSM Patrol Node — YASMIN-based Finite State Machine for Home Robot.
4
5This module implements a modular waypoint patrol system using YASMIN
6state machines. It wraps Nav2's NavigateToPose action inside YASMIN
7states, providing a visual FSM representation through the YASMIN Viewer.
8
9The FSM has the following structure:
10
11 ┌──────────────┐
12 │ SELECT_WP │──(finished)──► [patrol_done]
13 └──────┬───────┘
14 │ navigate
15 ┌──────▼───────┐
16 │ NAVIGATE │──(abort/cancel)──► HANDLE_ERROR
17 └──────┬───────┘
18 │ succeeded
19 ┌──────▼───────┐
20 │ WP_REACHED │──(next)──► SELECT_WP
21 └──────────────┘
22
23 ┌──────────────┐
24 │ HANDLE_ERROR │──(recover)──► SELECT_WP
25 │ │──(abort)────► [patrol_done]
26 └──────────────┘
27
28Usage:
29 # After launching the navigation stack (main_nav.launch.py):
30 ros2 run home_robot_fsm fsm_patrol
31 ros2 run home_robot_fsm fsm_patrol -- --random
32 ros2 run home_robot_fsm fsm_patrol -- --random --skip-errors
33
34 # Or via launch file (starts viewer automatically):
35 ros2 launch home_robot_fsm fsm_patrol.launch.py
36 ros2 launch home_robot_fsm fsm_patrol.launch.py random:=true
37
38Note:
39 This package does NOT modify any existing home_robot code.
40 It reads the waypoints from the home_robot package's installed
41 config and uses Nav2's action interface independently.
42"""

72def load_waypoints() -> dict:
73 """Load waypoints from the home_robot package's waypoints.yaml.
74
75 Tries the installed share directory first, falls back to
76 PROJECT_ROOT environment variable.
77
78 Returns:
79 dict: Waypoint dictionary ``{name: {x, y, theta}}``.
80
81 Raises:
82 FileNotFoundError: If the waypoints file cannot be located.
83 """
84 # Primary: installed package path
85 try:
86 pkg_share = get_package_share_directory('home_robot')
87 wp_path = os.path.join(pkg_share, 'config', 'waypoints.yaml')
88 except Exception:
89 # Fallback: PROJECT_ROOT env var (Docker default)
90 project_root = os.environ.get(
91 'PROJECT_ROOT', '/home/ubuntu/home_robot')
92 wp_path = os.path.join(
93 project_root, 'ros2_ws/src/home_robot/config/waypoints.yaml')
94
95 if not os.path.exists(wp_path):
96 raise FileNotFoundError(
97 f'Waypoints file not found at {wp_path}')
98
99 with open(wp_path, 'r') as f:
100 data = yaml.safe_load(f)
101 return data.get('waypoints', {})

105# State 1: SELECT_WAYPOINT — picks the next waypoint

107
108class SelectWaypointState(State):
109 """Selects the next waypoint from the queue and writes it to the
110 blackboard.
111
112 Outcomes:
113 ``navigate`` — a waypoint was selected, proceed to navigation.
114 ``finished`` — all waypoints have been visited.
115 """
116
117 def __init__(self) -> None:
118 super().__init__(['navigate', 'finished'])
119
120 def execute(self, blackboard: Blackboard) -> str:
121 wp_names = blackboard['wp_names']
122 wp_dict = blackboard['wp_dict']
123 wp_index = int(blackboard['wp_index'])
124
125 if wp_index >= len(wp_names):
126 yasmin.YASMIN_LOG_INFO(
127 '🏁 All waypoints visited! Patrol complete.')
128 return 'finished'
129
130 # Advance by index instead of mutating a list in-place. This makes the
131 # current patrol position explicit in the blackboard across transitions.
132 wp_name = wp_names[wp_index]
133 blackboard['wp_index'] = wp_index + 1
134 wp_data = wp_dict[wp_name]
135
136 # Store in blackboard for the NavigateState
137 blackboard['current_wp_name'] = wp_name
138 blackboard['current_wp_x'] = float(wp_data['x'])
139 blackboard['current_wp_y'] = float(wp_data['y'])
140 blackboard['current_wp_theta'] = float(wp_data['theta'])
141
142 total = blackboard['wp_total']
143 idx = wp_index + 1
144 yasmin.YASMIN_LOG_INFO(
145 f'[{idx}/{total}] 🎯 Selected waypoint: {wp_name} '
146 f'(x={wp_data["x"]}, y={wp_data["y"]})')
147
148 time.sleep(1.0)
149 return 'navigate'

152# ──────────────────────────────────────────────────────────────────────
153# State 2: NAVIGATE — ActionState wrapping NavigateToPose
154# ──────────────────────────────────────────────────────────────────────
155
156class NavigateState(ActionState):
157 """Sends a NavigateToPose goal to Nav2 and monitors progress.
158
159 Inherits from ``yasmin_ros.ActionState`` to handle the full
160 action lifecycle (goal / feedback / result) automatically.
161
162 Outcomes (inherited + custom):
163 ``succeeded`` — navigation completed successfully.
164 ``aborted`` — Nav2 aborted the goal.
165 ``canceled`` — navigation was canceled externally.
166 """
167
168 def __init__(self) -> None:
169 super().__init__(
170 NavigateToPose, # action type
171 '/navigate_to_pose', # action name
172 self._create_goal, # goal callback
173 set(), # extra outcomes (SUCCEED/ABORT/CANCEL auto)
174 self._handle_result, # result callback
175 self._handle_feedback, # feedback callback
176 )
177 self.last_dist = None
178 self.last_time = None
179
180 def _create_goal(
181 self, blackboard: Blackboard
182 ) -> NavigateToPose.Goal:
183 """Build a NavigateToPose goal from blackboard data."""
184
185 # Reset stuck checking variables for this new goal
186 self.last_dist = None
187 self.last_time = None
188 blackboard['is_stuck'] = False
189
190 goal = NavigateToPose.Goal()
191 goal.pose = PoseStamped()
192 goal.pose.header.frame_id = 'map'
193
194 goal.pose.pose.position.x = blackboard['current_wp_x']
195 goal.pose.pose.position.y = blackboard['current_wp_y']
196
197 theta = blackboard['current_wp_theta']
198 goal.pose.pose.orientation.z = math.sin(theta / 2.0)
199 goal.pose.pose.orientation.w = math.cos(theta / 2.0)
200
201 yasmin.YASMIN_LOG_INFO(
202 f'📤 Sending goal to Nav2: {blackboard["current_wp_name"]}')
203 return goal
204
205 def _handle_result(
206 self,
207 blackboard: Blackboard,
208 result: NavigateToPose.Result,
209 ) -> str:
210 """Process the action result."""
211 yasmin.YASMIN_LOG_INFO(
212 f'✅ Nav2 reports success for: '
213 f'{blackboard["current_wp_name"]}')
214 return SUCCEED
215
216 def _handle_feedback(
217 self,
218 blackboard: Blackboard,
219 feedback: NavigateToPose.Feedback,
220 ) -> None:
221 """Log navigation feedback (remaining distance) and check for stuck."""
222 dist = feedback.distance_remaining
223 current_time = time.time()
224
225 # Stuck detection logic
226 if self.last_dist is not None:
227 if abs(dist - self.last_dist) <= 0.1:
228 # Has been within tolerance for how long?
229 if (current_time - self.last_time) >= 15.0:
230 yasmin.YASMIN_LOG_WARN("🤖 Robot is physically stuck! Canceling navigation to recover.")
231 blackboard['is_stuck'] = True
232 self.cancel_state()
233 else:
234 # Moved out of tolerance, reset
235 self.last_dist = dist
236 self.last_time = current_time
237 else:
238 self.last_dist = dist
239 self.last_time = current_time
240
241 if dist > 0:
242 yasmin.YASMIN_LOG_INFO(
243 f' 📍 {blackboard["current_wp_name"]}: '
244 f'{dist:.2f} m remaining')

251class WaypointReachedState(State):
252 """Announces that the robot has reached a waypoint.
253
254 Publishes an event on ``/patrol_events`` so that the existing
255 ``voice_controller`` node (if running) can announce the arrival.
256
257 Outcomes:
258 ``next`` — ready to select the next waypoint.
259 """
260
261 def __init__(self, node: Node) -> None:
262 super().__init__(['next'])
263 self._pub = node.create_publisher(String, 'patrol_events', 10)
264
265 def execute(self, blackboard: Blackboard) -> str:
266 wp_name = blackboard['current_wp_name']
267 yasmin.YASMIN_LOG_INFO(
268 f'🏠 Arrived at: {wp_name}')
269
270 # Add a sleep so the web viewer can actually catch this state being active!
271 time.sleep(1.5)
272
273 # Publish arrival event (compatible with voice_controller)
274 msg = String()
275 msg.data = f'llegada:{wp_name}'
276 self._pub.publish(msg)
277 return 'next'

296class StuckRecoveryState(State):
297 """Executes a recovery maneuver (moves backwards)."""
298 def __init__(self, node: Node) -> None:
299 super().__init__(['recovered'])
300 self.cmd_pub = node.create_publisher(Twist, '/cmd_vel', 10)
301
302 def execute(self, blackboard: Blackboard) -> str:
303 yasmin.YASMIN_LOG_WARN("🚧 Executing stuck recovery (moving backwards 1m/s for 1s)")
304
305 # Publish Twist backwards
306 twist = Twist()
307 twist.linear.x = -1.0
308
309 # Wait for the viewer to update visually
310 time.sleep(1.0)
311
312 self.cmd_pub.publish(twist)
313 time.sleep(1.2) # Give it a bit over 1s backwards
314
315 # Stop
316 twist.linear.x = 0.0
317 self.cmd_pub.publish(twist)
318 time.sleep(0.5)
319
320 return 'recovered'

327class HandleErrorState(State):
328 """Handles navigation failures.
329
330 If ``skip_errors`` is enabled in the blackboard, the state
331 transitions to *recover* (try the next waypoint). Otherwise
332 it aborts the patrol.
333
334 Outcomes:
335 ``recover`` — skip this waypoint, try the next one.
336 ``abort`` — stop the patrol entirely.
337 """
338
339 def __init__(self) -> None:
340 super().__init__(['recover', 'abort'])
341
342 def execute(self, blackboard: Blackboard) -> str:
343 wp_name = blackboard['current_wp_name'] if 'current_wp_name' in blackboard else '???'
344 skip = blackboard['skip_errors'] if 'skip_errors' in blackboard else False
345
346 if skip:
347 yasmin.YASMIN_LOG_WARN(
348 f'⚠ Navigation failed for {wp_name}, skipping...')
349 return 'recover'
350 else:
351 yasmin.YASMIN_LOG_ERROR(
352 f'❌ Navigation failed for {wp_name}, aborting patrol.')
353 return 'abort'

356# ──────────────────────────────────────────────────────────────────────
357# Build the State Machine
358# ──────────────────────────────────────────────────────────────────────
359
360def build_fsm(node: Node) -> StateMachine:
361 """Construct and return the patrol FSM.
362
363 Args:
364 node: ROS 2 node used for publishers inside states.
365
366 Returns:
367 StateMachine: The fully wired YASMIN state machine.
368 """
369 sm = StateMachine(outcomes=['patrol_done'], handle_sigint=True)
370
371 sm.add_state(
372 'SELECT_WAYPOINT',
373 SelectWaypointState(),
374 transitions={
375 'navigate': 'NAVIGATE',
376 'finished': 'patrol_done',
377 },
378 )
379
380 sm.add_state(
381 'NAVIGATE',
382 NavigateState(),
383 transitions={
384 SUCCEED: 'WAYPOINT_REACHED',
385 ABORT: 'HANDLE_ERROR',
386 CANCEL: 'CHECK_STUCK',
387 },
388 )
389
390 sm.add_state(
391 'CHECK_STUCK',
392 CheckStuckState(),
393 transitions={
394 'stuck': 'STUCK_RECOVERY',
395 'error': 'HANDLE_ERROR',
396 },
397 )
398
399 sm.add_state(
400 'STUCK_RECOVERY',
401 StuckRecoveryState(node),
402 transitions={
403 'recovered': 'NAVIGATE',
404 },
405 )
406
407 sm.add_state(
408 'WAYPOINT_REACHED',
409 WaypointReachedState(node),
410 transitions={
411 'next': 'SELECT_WAYPOINT',
412 },
413 )
414
415 sm.add_state(
416 'HANDLE_ERROR',
417 HandleErrorState(),
418 transitions={
419 'recover': 'SELECT_WAYPOINT',
420 'abort': 'patrol_done',
421 },
422 )
423
424 return sm

431def main() -> None:
432 """Entry point for the FSM patrol node.
433
434 CLI flags (pass after ``--``):
435 ``--random`` Shuffle waypoints instead of sequential order.
436 ``--skip-errors`` Skip failed waypoints instead of aborting.
437 """
438 rclpy.init()
439
440 # ROS 2 + YASMIN logging
441 set_ros_loggers()
442 yasmin.YASMIN_LOG_INFO('🤖 home_robot_fsm — FSM Patrol starting')
443
444 # Create a lightweight node for publishers
445 node = rclpy.create_node('fsm_patrol_node')
446
447 # Parse CLI flags
448 use_random = '--random' in sys.argv
449 skip_errors = '--skip-errors' in sys.argv
450
451 # Load waypoints from home_robot package
452 try:
453 wp_dict = load_waypoints()
454 except FileNotFoundError as e:
455 yasmin.YASMIN_LOG_ERROR(str(e))
456 rclpy.shutdown()
457 return
458
459 wp_names = list(wp_dict.keys())
460 if use_random:
461 random_module.shuffle(wp_names)
462 yasmin.YASMIN_LOG_INFO('🎲 Random mode activated')
463 else:
464 yasmin.YASMIN_LOG_INFO('📋 Sequential mode')
465
466 yasmin.YASMIN_LOG_INFO(
467 f'Loaded {len(wp_names)} waypoints: {", ".join(wp_names)}')
468
469 # Build FSM
470 sm = build_fsm(node)
471
472 # Publish FSM structure for YASMIN Viewer
473 YasminViewerPub(sm, 'HOME_ROBOT_FSM_PATROL')
474
475 # Prepare blackboard
476 blackboard = Blackboard()
477 blackboard['wp_dict'] = wp_dict
478 blackboard['wp_names'] = list(wp_names)
479 blackboard['wp_index'] = 0
480 blackboard['wp_total'] = len(wp_names)
481 blackboard['skip_errors'] = skip_errors
482
483 # Execute FSM
484 try:
485 outcome = sm(blackboard)
486 yasmin.YASMIN_LOG_INFO(f'FSM finished with outcome: {outcome}')
487 except KeyboardInterrupt:
488 yasmin.YASMIN_LOG_WARN('🛑 Patrol interrupted by user')
489 except Exception as e:
490 yasmin.YASMIN_LOG_ERROR(f'FSM error: {e}')
491 finally:
492 node.destroy_node()
493 if rclpy.ok():
494 rclpy.shutdown()
495
496
497if __name__ == '__main__':
498 main()
